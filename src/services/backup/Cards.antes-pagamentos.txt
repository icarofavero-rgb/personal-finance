import { useState } from 'react'

import {
  getCards,
  saveCards,
  type CreditCard,
} from '../services/cardsStorage'

import {
  getCardPurchases,
  saveCardPurchases,
  type CardPurchase,
} from '../services/cardPurchasesStorage'

import {
  calculateInstallments,
  getCardInvoices,
} from '../services/cardInvoices'

function calculateUsedLimit(
  cardId: string,
  purchases: CardPurchase[]
): number {
  return purchases
    .filter(purchase => purchase.cardId === cardId)
    .reduce((total, purchase) => total + purchase.amount, 0)
}

const formatMoney = (value: number) =>
  new Intl.NumberFormat('pt-BR', {
    style: 'currency',
    currency: 'BRL',
  }).format(value)

export default function Cards() {
  const [cards, setCards] = useState<CreditCard[]>(getCards)
  const [purchases, setPurchases] =
  useState<CardPurchase[]>(getCardPurchases)
  const invoices = getCardInvoices(purchases, cards)

  const [purchaseCardId, setPurchaseCardId] = useState('')
  const [purchaseDescription, setPurchaseDescription] = useState('')
  const [purchaseAmount, setPurchaseAmount] = useState('')
  const [purchaseCategory, setPurchaseCategory] = useState('Outros')
  const [purchaseDate, setPurchaseDate] = useState('')
  const [purchaseInstallments, setPurchaseInstallments] = useState('1')

  const [name, setName] = useState('')
  const [bank, setBank] = useState('')
  const [limit, setLimit] = useState('')
  const [closingDay, setClosingDay] = useState('10')
  const [dueDay, setDueDay] = useState('17')
  
function handleAddPurchase(event: React.FormEvent<HTMLFormElement>) {
  event.preventDefault()

  const amount = Number(purchaseAmount)
  const installments = Number(purchaseInstallments)

  if (!cards.some(card => card.id === purchaseCardId)) {
    alert('Selecione um cartão válido.')
    return
  }

  if (
    !purchaseDescription.trim() ||
    !Number.isFinite(amount) ||
    amount <= 0 ||
    !Number.isInteger(installments) ||
    installments < 1 ||
    installments > 120 ||
    !/^\d{4}-\d{2}-\d{2}$/.test(purchaseDate)
  ) {
    alert('Confira os dados da compra.')
    return
  }

  const newPurchase: CardPurchase = {
    id: crypto.randomUUID(),
    cardId: purchaseCardId,
    description: purchaseDescription.trim(),
    amount,
    category: purchaseCategory,
    date: purchaseDate,
    installments,
  }

  try {
    const updated = [...purchases, newPurchase]

    saveCardPurchases(updated)
    setPurchases(updated)

    setPurchaseDescription('')
    setPurchaseAmount('')
    setPurchaseCategory('Outros')
    setPurchaseDate('')
    setPurchaseInstallments('1')

    alert('Compra cadastrada com sucesso!')
  } catch {
    alert('Não foi possível salvar a compra.')
  }
}

  function handleAddCard(event: React.FormEvent<HTMLFormElement>) {
    event.preventDefault()

    const parsedLimit = Number(limit)
    const closing = Number(closingDay)
    const due = Number(dueDay)

    if (
      !name.trim() ||
      !bank.trim() ||
      !Number.isFinite(parsedLimit) ||
      parsedLimit <= 0 ||
      !Number.isInteger(closing) ||
      closing < 1 ||
      closing > 31 ||
      !Number.isInteger(due) ||
      due < 1 ||
      due > 31
    ) {
      alert('Confira os dados do cartão.')
      return
    }

    const newCard: CreditCard = {
      id: crypto.randomUUID(),
      name: name.trim(),
      bank: bank.trim(),
      limit: parsedLimit,
      closingDay: closing,
      dueDay: due,
    }

    const updated = [...cards, newCard]

    setCards(updated)
    saveCards(updated)

    setName('')
    setBank('')
    setLimit('')
    setClosingDay('10')
    setDueDay('17')
  }

  function handleDeleteCard(id: string) {
    if (!window.confirm('Deseja excluir este cartão?')) {
      return
    }

    const updated = cards.filter(card => card.id !== id)
    setCards(updated)
    saveCards(updated)
  }

  return (
    <div className="cards-page">
      <header className="reports-header">
        <div>
          <span className="reports-eyebrow">
            GESTÃO FINANCEIRA
          </span>
          <h1>Cartões de crédito</h1>
          <p>Gerencie seus cartões e seus limites.</p>
        </div>
      </header>

      <section className="panel">
        <h3>Cadastrar novo cartão</h3>

        <form onSubmit={handleAddCard} className="cards-form">
          <label>
            Nome do cartão
            <input
              value={name}
              onChange={event => setName(event.target.value)}
              placeholder="Ex.: Nubank Platinum"
              required
            />
          </label>

          <label>
            Instituição financeira
            <input
              value={bank}
              onChange={event => setBank(event.target.value)}
              placeholder="Ex.: Nubank"
              required
            />
          </label>

          <label>
            Limite total (R$)
            <input
              type="number"
              min="0.01"
              step="0.01"
              value={limit}
              onChange={event => setLimit(event.target.value)}
              required
            />
          </label>

          <label>
            Dia do fechamento
            <input
              type="number"
              min="1"
              max="31"
              value={closingDay}
              onChange={event => setClosingDay(event.target.value)}
              required
            />
          </label>

          <label>
            Dia do vencimento
            <input
              type="number"
              min="1"
              max="31"
              value={dueDay}
              onChange={event => setDueDay(event.target.value)}
              required
            />
          </label>

          <button type="submit" className="primary-button">
            Adicionar cartão
          </button>
        </form>
      </section>

<section className="panel">
  <h3>Registrar compra no cartão</h3>

  <form onSubmit={handleAddPurchase} className="purchase-form">
    <label>
      Cartão
      <select
        value={purchaseCardId}
        onChange={event => setPurchaseCardId(event.target.value)}
        required
      >
        <option value="">Selecione um cartão</option>
        {cards.map(card => (
          <option key={card.id} value={card.id}>
            {card.name} — {card.bank}
          </option>
        ))}
      </select>
    </label>

    <label>
      Descrição
      <input
        value={purchaseDescription}
        onChange={event => setPurchaseDescription(event.target.value)}
        placeholder="Ex.: Supermercado"
        required
      />
    </label>

    <label>
      Valor total (R$)
      <input
        type="number"
        min="0.01"
        step="0.01"
        value={purchaseAmount}
        onChange={event => setPurchaseAmount(event.target.value)}
        required
      />
    </label>

    <label>
      Categoria
      <select
        value={purchaseCategory}
        onChange={event => setPurchaseCategory(event.target.value)}
      >
        <option value="Outros">Outros</option>
        <option value="Alimentação">Alimentação</option>
        <option value="Transporte">Transporte</option>
        <option value="Moradia">Moradia</option>
        <option value="Lazer">Lazer</option>
        <option value="Saúde">Saúde</option>
        <option value="Compras">Compras</option>
      </select>
    </label>

    <label>
      Data da compra
      <input
        type="date"
        value={purchaseDate}
        onChange={event => setPurchaseDate(event.target.value)}
        required
      />
    </label>

    <label>
      Quantidade de parcelas
      <input
        type="number"
        min="1"
        max="120"
        step="1"
        value={purchaseInstallments}
        onChange={event => setPurchaseInstallments(event.target.value)}
        required
      />
    </label>

    <button type="submit" className="primary-button">
      Registrar compra
    </button>
  </form>
</section>


<section className="panel">
  <h3>Compras no cartão</h3>

  {purchases.length === 0 ? (
    <p>Nenhuma compra registrada ainda.</p>
  ) : (
    <div className="cards-list">
      {purchases.map(purchase => {
        const card = cards.find(
          item => item.id === purchase.cardId
        )

        return (
          <div className="purchase-item" key={purchase.id}>
            <h3>{purchase.description}</h3>

            <p>
              Cartão: {card?.name ?? 'Cartão não encontrado'}
            </p>

            <p>Valor total: {formatMoney(purchase.amount)}</p>

            <p>Categoria: {purchase.category}</p>

            <p>Data: {purchase.date}</p>

            <p>
              Parcelamento: {purchase.installments}x
            </p>

          {card && (
  <div className="purchase-installments">
    <h4>Parcelas por fatura</h4>

    {calculateInstallments(purchase, card).map(installment => (
      <div
        className="purchase-installment-row"
        key={`${installment.purchaseId}-${installment.installmentNumber}`}
      >
        <span>
          {installment.invoiceMonth} — Parcela{' '}
          {installment.installmentNumber}/
          {installment.totalInstallments}
        </span>

        <strong>{formatMoney(installment.amount)}</strong>
      </div>
    ))}
  </div>
)}
          </div>
        )
      })}
    </div>
  )}
</section>


<section className="panel">
  <h3>Minhas faturas</h3>

  {invoices.length === 0 ? (
    <p>Nenhuma fatura encontrada.</p>
  ) : (
    <div className="invoices-grid">
      {invoices.map(invoice => {
        const card = cards.find(
          item => item.id === invoice.cardId
        )

        return (
          <article
            className="invoice-item"
            key={`${invoice.cardId}-${invoice.invoiceMonth}`}
          >
            <div className="invoice-header">
              <div>
                <small>{card?.name ?? 'Cartão não encontrado'}</small>
                <h4>Fatura de {invoice.invoiceMonth}</h4>
              </div>

              <strong>{formatMoney(invoice.total)}</strong>
            </div>

            <div className="invoice-installments">
              {invoice.installments.map(installment => (
                <div
                  className="invoice-installment-row"
                  key={`${installment.purchaseId}-${installment.installmentNumber}`}
                >
                  <span>
                    {installment.description} (
                    {installment.installmentNumber}/
                    {installment.totalInstallments})
                  </span>

                  <strong>
                    {formatMoney(installment.amount)}
                  </strong>
                </div>
              ))}
            </div>

            <p className="invoice-note">
              Pagamento ainda não implementado.
            </p>
          </article>
        )
      })}
    </div>
  )}
</section>

      <section className="cards-list">
        <h3>Meus cartões</h3>

<p>
  Compras registradas: {purchases.length}
</p>

        {cards.length === 0 ? (
          <p>Nenhum cartão cadastrado ainda.</p>
        ) : (
          <div className="cards-grid">
            {cards.map(card => (
              <article className="credit-card-item" key={card.id}>
                <div className="credit-card-top">
                  <strong>{card.name}</strong>
                  <span>{card.bank}</span>
                </div>

                <div className="credit-card-limit">
  <small>Limite total</small>
  <h2>{formatMoney(card.limit)}</h2>

  <p>
    Limite utilizado:{' '}
    {formatMoney(calculateUsedLimit(card.id, purchases))}
  </p>

  <p>
    Limite disponível:{' '}
    {formatMoney(
      card.limit - calculateUsedLimit(card.id, purchases)
    )}
  </p>
</div>

                <div className="credit-card-dates">
                  <span>Fechamento: dia {card.closingDay}</span>
                  <span>Vencimento: dia {card.dueDay}</span>
                </div>

                <button
                  type="button"
                  className="card-delete-button"
                  onClick={() => handleDeleteCard(card.id)}
                >
                  Excluir cartão
                </button>
              </article>
            ))}
          </div>
        )}
      </section>
    </div>
  )
}
