
import type { CreditCard } from './cardsStorage'
import type { CardInvoice } from './cardInvoices'
import type { CardPayment } from './cardPaymentsStorage'
import type { Transaction } from './storage'

export type PaymentValidationResult = {
  valid: boolean
  error?: string
}

function isValidDate(value: string): boolean {
  if (!/^\d{4}-\d{2}-\d{2}$/.test(value)) {
    return false
  }

  const [year, month, day] = value.split('-').map(Number)
  const date = new Date(Date.UTC(year, month - 1, day))

  return (
    date.getUTCFullYear() === year &&
    date.getUTCMonth() + 1 === month &&
    date.getUTCDate() === day
  )
}

export function validateCardPayment(
  payment: CardPayment,
  cards: CreditCard[],
  invoices: CardInvoice[],
  existingPayments: CardPayment[],
  existingTransactions: Transaction[] = []
): PaymentValidationResult {
  if (
    !payment.id?.trim() ||
    !payment.cardId?.trim() ||
    !payment.invoiceMonth?.trim() ||
    !payment.date?.trim()
  ) {
    return {
      valid: false,
      error: 'Existem campos obrigatórios não preenchidos.',
    }
  }

  if (
    !/^\d{4}-(0[1-9]|1[0-2])$/.test(payment.invoiceMonth) ||
    !isValidDate(payment.date)
  ) {
    return {
      valid: false,
      error: 'Data ou mês da fatura inválido.',
    }
  }

  const amountCents = Math.round(payment.amount * 100)

  if (
    !Number.isFinite(payment.amount) ||
    payment.amount <= 0 ||
    !Number.isSafeInteger(amountCents) ||
    Math.abs(payment.amount * 100 - amountCents) > 0.000001
  ) {
    return {
      valid: false,
      error: 'Informe um valor válido com até duas casas decimais.',
    }
  }

  if (
    existingPayments.some(item => item.id === payment.id)
  ) {
    return {
      valid: false,
      error: 'Este pagamento já foi registrado.',
    }
  }

  if (payment.transactionId !== undefined) {
    return {
      valid: false,
      error: 'O vínculo da transação deve ser criado pelo sistema.',
    }
  }

  const cardExists = cards.some(
    card => card.id === payment.cardId
  )

  if (!cardExists) {
    return {
      valid: false,
      error: 'O cartão informado não existe.',
    }
  }

  const invoiceExists = invoices.some(
    invoice =>
      invoice.cardId === payment.cardId &&
      invoice.invoiceMonth === payment.invoiceMonth
  )

  if (!invoiceExists) {
    return {
      valid: false,
      error: 'A fatura informada não existe.',
    }
  }

  if (
    existingPayments.some(
      item =>
        item.transactionId !== undefined &&
        existingTransactions.some(
          transaction => transaction.id === item.transactionId
        ) === false
    )
  ) {
    return {
      valid: false,
      error: 'Existem pagamentos com vínculos de transação inconsistentes.',
    }
  }

  return { valid: true }
}
